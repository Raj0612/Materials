from Employee
left join (
    select team_id, count(*) as count
    from Employee
    group by team_id
) team 
on team.team_id = Employee.team_id

select w.name as warehouse_name, 
 sum(w.units * sub.vol) as volume
from Warehouse w
left join (
    select p.product_id, p.Width * p.Length * p.Height as vol 
    from Products p

) as sub 
on w.product_id = sub.product_id
group by warehouse_name;

 Number of Comments per Post
 https://leetcode.ca/2019-04-24-1241-Number-of-Comments-per-Post/
 https://github.com/Qian-Yu-2020/Leetcode-Problems/blob/master/Social%20Media%20and%20Posts/%23%201241.%20Number%20of%20Comments%20per%20Post.sql
 Write a solution to find the number of comments per post. The result table should contain post_id and its corresponding number_of_comments.

The Submissions table may contain duplicate comments. You should count the number of unique comments per post.

The Submissions table may contain duplicate posts. You should treat them as one post.

The result table should be ordered by post_id in ascending order.

 select s1.sub_id as post_id,
count(distinct s2.sub_id) as number_of_comments
from Submissions as s1
left join Submissions as s2
on s1.sub_id = s2.parent_id
where s1.parent_id is null
group by s1.sub_id
 
Percentage of Users Attended a Contest

select contest_id, 
round(count(distinct user_id) * 100 / (
    select count(user_id) from Users
), 2) as percentage
from Register
group by contest_id
order by percentage desc, contest_id

from Users u
left join Rides r
on u.id = r.user_id
group by u.id
order by travelled_distance desc, name asc

https://dev.to/ranggakd/average-selling-price-leetcode-mssql-37od
select p.product_id, 
ifnull(round( sum(u.units * p.price) / sum(units), 2), 0) as average_price
from Prices p 
left join UnitsSold u
on p.product_id = u.product_id and
 u.purchase_date between p.start_date and p.end_date
group by product_id
